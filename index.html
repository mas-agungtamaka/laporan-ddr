create table if not exists public.ddr_state (
  id text primary key,
  payload jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);
insert into public.ddr_state (id, payload)
values ('main_app', '{}'::jsonb)
on conflict (id) do nothing;
